-- INDEX v0.4 — relationships between entries -- Run in Supabase → SQL Editor → New snippet. create table if not exists public.index_relationships ( id bigint generated by default as identity primary key, record_code text not null references public.index_records(code) on delete cascade, entry_a bigint not null references public.index_entries(id) on delete cascade, entry_b bigint not null references public.index_entries(id) on delete cascade, relation_type text not null, participant_id text not null, created_at timestamptz not null default now(), constraint index_relationships_ordered_pair check (entry_a < entry_b), constraint index_relationships_type_values check ( relation_type in ( 'same_subject', 'same_moment', 'sequence', 'context', 'contrast' ) ), constraint index_relationships_one_per_source unique (entry_a, entry_b, relation_type, participant_id) ); create index if not exists index_relationships_record_idx on public.index_relationships (record_code, created_at); create index if not exists index_relationships_entry_a_idx on public.index_relationships (entry_a); create index if not exists index_relationships_entry_b_idx on public.index_relationships (entry_b); alter table public.index_relationships enable row level security; drop policy if exists "INDEX relationships are readable" on public.index_relationships; drop policy if exists "INDEX relationships can be proposed" on public.index_relationships; create policy "INDEX relationships are readable" on public.index_relationships for select to anon, authenticated using (true); create policy "INDEX relationships can be proposed" on public.index_relationships for insert to anon, authenticated with check ( entry_a < entry_b and char_length(participant_id) between 1 and 100 ); -- Confirm that both entries truly belong to the stated record. create or replace function public.validate_index_relationship() returns trigger language plpgsql security definer set search_path = public as $$ declare record_a text; record_b text; begin select record_code into record_a from public.index_entries where id = new.entry_a; select record_code into record_b from public.index_entries where id = new.entry_b; if record_a is null or record_b is null then raise exception 'Relationship entry not found'; end if; if record_a <> new.record_code or record_b <> new.record_code then raise exception 'Both entries must belong to the stated record'; end if; return new; end; $$; drop trigger if exists validate_index_relationship_trigger on public.index_relationships; create trigger validate_index_relationship_trigger before insert or update on public.index_relationships for each row execute function public.validate_index_relationship();
